FBFE: Scroll Window
Scrolls a window of the screen up one pixel line at a time, filling the new bottom line from ScrollWindow_Fill. Called by BASIC as "scr" (FBFE).
ScrollWindow FBFE DI Disable interrupts.
FBFF LD ($FF76),SP
FC03 LD SP,$FFBE
FC06 LD A,($FBF9)
FC09 LD ($FC44),A
ScrollWindow_0 FC0C LD A,($FBFB)
FC0F LD C,A
FC10 LD A,($FBFA)
FC13 LD B,A
FC14 CALL ScrollWindow_ScreenAddress
FC17 LD A,($FBFD)
FC1A SUB C
FC1B LD IYl,A
FC1D LD A,($FBFC)
FC20 SUB B
FC21 INC A
FC22 LD C,A
FC23 LD B,$00
ScrollWindow_1 FC25 LD D,H
FC26 LD E,L
FC27 CALL ScrollWindow_NextLine
FC2A LD A,L
FC2B LD IXl,C
FC2D LDIR
FC2F LD C,IXl
FC31 LD L,A
FC32 DEC IYl
FC34 JR NZ,ScrollWindow_1
FC36 CALL ScrollWindow_FillLine
FC39 LD HL,$FC44
FC3C DEC (HL)
FC3D JR NZ,ScrollWindow_0
FC3F LD SP,($FF76)
FC43 RET
FC44 EX AF,AF'
